@startuml
skinparam Monochrome true
skinparam StateBackgroundColor #F9F9F9
skinparam DefaultTextAlignment center

[*] --> LoggedIn : Buyer logs into account

state LoggedIn {
  [*] --> NavigatingOrderHistory : Navigates to order history
  NavigatingOrderHistory --> SelectingOrder : Selects order containing item
}

SelectingOrder --> DisplayingOrderItems : System displays associated items

DisplayingOrderItems --> SelectingItemForRefund : Buyer selects specific item
SelectingItemForRefund --> SelectingRefundOption : Selects refund request option

SelectingRefundOption --> CheckingEligibility : System validates refund eligibility

state CheckingEligibility <<choice>>
CheckingEligibility --> IneligibleItem : Item is not eligible [ALT2]
CheckingEligibility --> ProvidingReason : Item is eligible

state IneligibleItem {
  [*] --> NotifyingIneligible : System informs item not eligible
  NotifyingIneligible --> [*] : Prevents submission / Returns to order view
}

state ProvidingReason {
  [*] --> EnteringReason : Buyer provides reason for refund
  EnteringReason --> SubmittingRequest : Buyer submits refund request
}

SubmittingRequest --> PendingReview : System records RefundRequest & informs buyer

state PendingReview {
  [*] --> AwaitingDecision : Seller or administrator reviews request
  AwaitingDecision --> ReviewDecision : Decision made
}

state ReviewDecision <<choice>>
ReviewDecision --> RefundDenied : Request Denied [ALT1]
ReviewDecision --> RefundApproved : Request Approved

state RefundDenied {
  [*] --> NotifyingDenial : System informs buyer with reason
  NotifyingDenial --> [*] : End (Refund denied)
}

state RefundApproved {
  [*] --> NotifyingApproval : System informs buyer request approved
  NotifyingApproval --> [*] : End (Refund approved)
}

@enduml
