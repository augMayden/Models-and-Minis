@startuml
skinparam Monochrome true
skinparam StateBackgroundColor #F9F9F9
skinparam DefaultTextAlignment center

[*] --> BrowsingOrSearching : Buyer accesses webstore

state BrowsingOrSearching {
  [*] --> EnteringSearchQuery
  EnteringSearchQuery --> DisplayingResults : System displays listings from sellers
}

DisplayingResults --> SelectingListings : Buyer selects multiple listings to compare

state SelectingListings {
  [*] --> EvaluatingSelection
  EvaluatingSelection --> ComparingListings : Multiple listings selected
}

state ComparingListings {
  [*] --> DisplayingComparisonView
  DisplayingComparisonView --> InspectingAttributes : Compare price, description, seller, etc.
}

InspectingAttributes --> ValidatingListingSelection : Buyer selects preferred listing

state ValidatingListingSelection <<choice>>
ValidatingListingSelection --> ListingUnavailable : Listing is invalid or unavailable [ALT1]
ValidatingListingSelection --> ListingSelectedForPurchase : Listing is valid and available

state ListingUnavailable {
  [*] --> NotifyingAndRemoving : System informs buyer & removes from comparison
  NotifyingAndRemoving --> ComparingListings : Returns to comparison view
}

ListingSelectedForPurchase --> [*] : Proceeds to purchase flow

@enduml
